function v040dShuffle(list){
 const out=[...list];for(let i=out.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));const value=out[i];out[i]=out[j];out[j]=value;}return out;
}
function v040dCellKey(x,y){return x+','+y;}
function v040dEdgeKey(a,b){const left=a.key<b.key?a.key:b.key,right=a.key<b.key?b.key:a.key;return left+'|'+right;}
function v040dAdjacent(a,b){return Math.abs(a.gx-b.gx)+Math.abs(a.gy-b.gy)===1;}
function v040dBfs(cells,edges,start){
 const distances=new Map([[start.key,0]]),queue=[start];
 while(queue.length){const current=queue.shift(),distance=distances.get(current.key);for(const edge of edges){let next=null;if(edge.a===current.key)next=edge.b;else if(edge.b===current.key)next=edge.a;if(next&&!distances.has(next)){distances.set(next,distance+1);queue.push(cells.get(next));}}}
 return distances;
}

  const frontier=[];for(const cell of cells.values())for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){const gx=cell.gx+dx,gy=cell.gy+dy,key=v040dCellKey(gx,gy);if(gx<-2||gx>2||gy<0||gy>2||cells.has(key))continue;frontier.push({gx,gy,parent:cell});}

 }

 const degree=new Map();for(const cell of cells.values())degree.set(cell.key,0);for(const edge of edges){degree.set(edge.a,(degree.get(edge.a)||0)+1);degree.set(edge.b,(degree.get(edge.b)||0)+1);}const rooms=[...cells.values()].map(cell=>({id:cell.id,name:cell.id,x:cell.x,y:cell.y,w:V040D_ROOM_W,h:V040D_ROOM_H,tint:'#1d1819',combat:cell!==start&&cell!==exit,doors:cell.doors,v040dDistance:distances.get(cell.key)||0,v040dDegree:degree.get(cell.key)||0,v040dType:cell===start?'start':cell===exit?'exit':v040dWeighted(V040D_COMMON_TYPES),v040dRotation:Math.floor(Math.random()*4),v040dMinor:null,v040dRare:null,v040dScene:null,v040dBug:false}));

 return{rooms,corridors,edges,cells,startId:'start',exitId:'exit'};
}
